public class PlayerState : MonoBehaviour
{
    [SerializeField] private StateMachine _fsm;

    void Awake() => _fsm = new StateMachine(
        new IdleState(this),
        new MoveState(this),
        new DashState(this));

    void Update() => _fsm.Tick(Time.deltaTime);
}
